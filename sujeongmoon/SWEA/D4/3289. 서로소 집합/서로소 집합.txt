import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Solution {

	/**
	 *
	 * 문제 설명:
	 *
	 * 입력:
	 *
	 * 출력:

	 * 풀이방법:
	 *  n: 집합
	 m: 반복문 도는 줄

	 find(num)함수 -> int값 리턴, parents[num] 리턴
	 union(a, b) 함수 -> find(a) == find(b)해서 false나오면 parents[b] = a로 만들어줌

	 1. parents[n+1] 만들고, 1~n까지의 인덱스에 각각 자기의 값을 넣음 (자기 부모는 자신)
	 2. 0 ~~로 합집합 연산이 나오는 경우
	 -> a, b 중 값이 작은 걸 비교해서,
	 둘의 부모가 다른 경우 union함수로 parents[b] = a로 만들어줌
	 3. 1 ~~로 비교 연산이 나오는 경우,
	 -> find함수를 사용해 parents[a] == parents[b]인지 판단
	 -> 비교 뒤 sb를 활용해 0 or 1 추가
	 *
	 */

	static int n;
	static int m;
	static int[] parents;

	public static void main(String[] args) throws Exception {

		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();

		int T = Integer.parseInt(br.readLine());

		for (int t = 1; t <= T; t++) {

			StringTokenizer st = new StringTokenizer(br.readLine());

			n = Integer.parseInt(st.nextToken());
			m = Integer.parseInt(st.nextToken());

			parents = new int[n+1];

			for (int i = 1; i <= n; i++) {
				parents[i] = i;
			}

			sb.append("#").append(t).append(" ");

			for (int i = 0; i < m; i++) {
				st = new StringTokenizer(br.readLine());

				int num = Integer.parseInt(st.nextToken());
				int a = Integer.parseInt(st.nextToken());
				int b = Integer.parseInt(st.nextToken());

				if (num == 0) {
					union(a, b);
				} else {
					if (find(a) != find(b)) {
						sb.append(0);
					} else {
						sb.append(1);
					}
				}
			}

			sb.append("\n");
		}

		System.out.println(sb);
	}

	static int find(int num) {
		if (parents[num] == num) {
			return num;
		}
		return parents[num] = find(parents[num]);
	}

	static void union(int a, int b) {
		int max = Math.max(find(a), find(b));
		int min = Math.min(find(a),  find(b));

		if (max == min) {
			return; // a랑 b가 같은 경우 취급 주의
		}

		parents[max] = min;
		return;
	}



}
